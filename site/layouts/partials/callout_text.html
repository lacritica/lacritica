" - " " --- " }} {{ $html := $text | markdownify }} {{ $html = replace $html "—" "—" }} {{ $html = replace $html "—" "—" }} {{ $html = replace $html "–" "–" }} {{ $html = replace $html "–" "–" }} {{ return (safeHTML $html) }}
